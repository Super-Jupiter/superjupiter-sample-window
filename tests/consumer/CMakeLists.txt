cmake_minimum_required(VERSION 3.16)
project(SuperJupiterSampleWindowConsumer LANGUAGES CXX)
if(DEFINED SUPERJUPITER_SAMPLE_WINDOW_SOURCE_DIR)
  add_subdirectory("${SUPERJUPITER_SAMPLE_WINDOW_SOURCE_DIR}" sample-window)
  if(TARGET superjupiter-sample-window-tests OR TARGET superjupiter-sample-window-replay)
    message(FATAL_ERROR "A consumer unexpectedly builds library developer tools")
  endif()
else()
  find_package(SuperJupiterSampleWindow 0.1 CONFIG REQUIRED)
endif()
add_executable(superjupiter-sample-window-consumer main.cpp)
target_link_libraries(superjupiter-sample-window-consumer PRIVATE SuperJupiter::SampleWindow)
